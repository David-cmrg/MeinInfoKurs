        int Jahre = 0;

        System.out.println("Zu beginn der Simulation leben " + AktuelleBewohner + " Fische im Teich.");

        while (AktuelleBewohner < (MaximalEinwohner - 0.01)) {
            x = (MaximalEinwohner - AktuelleBewohner) * Reproduktionsrate + AktuelleBewohner;
            AktuelleBewohner = x;
            System.out.println("    Aktuelle Bewohnerzahl" + " (Jahr: " + Jahre + "): " + InOut.format2(AktuelleBewohner));
            Jahre++;
